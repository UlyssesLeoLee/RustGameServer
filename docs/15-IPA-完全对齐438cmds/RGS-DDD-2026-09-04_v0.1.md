```rust
// 5 域 error.rs 通用模式 (per player-service error.rs 9KB)
use thiserror::Error;

#[derive(Debug, Error)]
pub enum Error {
    #[error("database error: {0}")]
    Database(#[from] sqlx::Error),
    #[error("not found: {entity} with id {id}")]
    NotFound { entity: String, id: String },
    #[error("validation: {0}")]
    Validation(String),
    #[error("nickname taken: {0}")]
    NicknameTaken(String),
    #[error("session expired")]
    SessionExpired,
    #[error("account disabled: {0}")]
    AccountDisabled(String),
    #[error("forbidden: {0}")]
    Forbidden(String),
    #[error("internal error: {0}")]
    Internal(#[from] anyhow::Error),
}

impl From<Error> for tonic::Status {
    fn from(err: Error) -> Self {
        match err {
            Error::NotFound { .. } => Status::not_found(err.to_string()),
            Error::Validation(_) | Error::NicknameTaken(_) => Status::invalid_argument(err.to_string()),
            Error::Forbidden(_) | Error::AccountDisabled(_) => Status::permission_denied(err.to_string()),
            Error::SessionExpired => Status::unauthenticated(err.to_string()),
            _ => Status::internal(err.to_string()),
        }
    }
}
```
