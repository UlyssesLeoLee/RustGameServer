export function getState() { return state; }
export function setState(patch) {
  state = { ...state, ...patch };
  subscribers.forEach(fn => fn(state));

export function subscribe(fn) {
  subscribers.add(fn);
  return () => subscribers.delete(fn);
}

### 2.3 data-loader.js

```js
// src/data-loader.js
export async function loadAll({ baseUrl = '' } = {}) {
  const [data, categories, i18n] = await Promise.all([
    fetchJson(`${baseUrl}data/api.json`),
    fetchJson(`${baseUrl}data/categories.json`),
    fetchJson(`${baseUrl}data/i18n.json`)
  ]);
  validate(data, categories);
  return { data, categories, i18n };
}

async function fetchJson(url) {
  const res = await fetch(url, { cache: 'no-cache' });
  if (!res.ok) throw new Error(`HTTP ${res.status} loading ${url}`);
  return res.json();
}

function validate(data, categories) {
  if (!Array.isArray(data.apis)) throw new Error('api.json: apis must be array');
  for (const api of data.apis) {
    if (!api.id) throw new Error('api.json: api.id missing');
    if (!categories.categories[api.primaryEngine]) {
      throw new Error(`api.json: unknown engine ${api.primaryEngine} (api: ${api.id})`);
    }
    if (!categories.categories[api.primaryEngine].find(c => c.id === api.category)) {
      throw new Error(`api.json: unknown category ${api.category} for ${api.primaryEngine} (api: ${api.id})`);
    }
  }
}

// src/list.js
export function getFilteredApis(state) {
  let apis = state.data.apis;
  if (state.engine !== 'all') {
    apis = apis.filter(a => a.primaryEngine === state.engine);
  }
  if (state.category !== 'all') {
    apis = apis.filter(a => a.category === state.category);
  }
  if (state.search) {
    const q = state.search.toLowerCase();
    apis = apis.filter(a =>
      a.name.toLowerCase().includes(q) ||
      a.summary.toLowerCase().includes(q) ||
      (a.tags || []).some(t => t.toLowerCase().includes(q))
    );
  }
  return apis;
}

export function renderList(state, container) {

// src/modal.js
export function openModal(apiId) {
  const api = getState().data.apis.find(a => a.id === apiId);
  if (!api) return;
  setState({ selectedApiId: apiId, modalTab: 'overview' });
  document.getElementById('modal-root').setAttribute('aria-hidden', 'false');
  trapFocus();
}

export function closeModal() {
  setState({ selectedApiId: null });
  document.getElementById('modal-root').setAttribute('aria-hidden', 'true');
  releaseFocus();
}

function trapFocus() {
  const modal = document.getElementById('modal');
  const focusable = modal.querySelectorAll('button, [tabindex="0"]');
  if (focusable[0]) focusable[0].focus();
  modal.addEventListener('keydown', onKeydown);
}

function onKeydown(e) {
  if (e.key === 'Escape') { closeModal(); return; }
  if (e.key === 'Tab') {
    // 标准焦点循环
    const focusable = [...e.currentTarget.querySelectorAll('button, [tabindex="0"]')];
    const first = focusable[0];
    const last = focusable[focusable.length - 1];
    if (e.shiftKey && document.activeElement === first) {
      e.preventDefault(); last.focus();
    } else if (!e.shiftKey && document.activeElement === last) {
      e.preventDefault(); first.focus();
    }
  }
}

export function renderModal(state, container) {
  if (!state.selectedApiId) { container.innerHTML = ''; return; }
  const api = state.data.apis.find(a => a.id === state.selectedApiId);
  const tabs = ['overview', 'params', 'examples', 'caveats', 'diff'];

  container.innerHTML = `
    <div class="modal-overlay" data-close="1"></div>
    <div class="modal" role="dialog" aria-modal="true" aria-labelledby="modal-title">
      <header class="modal-header">
        <h2 id="modal-title">${escapeHtml(api.name)}</h2>
        <button class="modal-close" data-close="1" aria-label="${state.i18n.strings['modal.close']}">×</button>
      </header>
      <div class="modal-meta">
        <span class="engine-badge engine-${api.primaryEngine}">${api.primaryEngine}</span>
        <span class="signature">${escapeHtml(api.signature)}</span>
        ${(api.tags || []).map(t => `<span class="tag">${escapeHtml(t)}</span>`).join('')}
      </div>
      <nav class="modal-tabs" role="tablist">
        ${tabs.map(t => `<button role="tab" class="modal-tab ${state.modalTab === t ? 'active' : ''}"
                            data-modal-tab="${t}">${state.i18n.strings[`tab.${t === 'diff' ? 'diff' : t}`]}</button>`).join('')}
      </nav>
      <div class="modal-body">${renderModalTabContent(state, api)}</div>
    </div>
  `;
}

function renderModalTabContent(state, api) {
  switch (state.modalTab) {
    case 'overview': return `<p>${escapeHtml(api.description)}</p>`;
    case 'params':   return renderParams(api);
    case 'examples': return renderExamples(state, api);
    case 'caveats':  return `<ul>${(api.caveats || []).map(c => `<li>${escapeHtml(c)}</li>`).join('')}</ul>`;
    case 'diff':     return renderEngineDiff(api);
  }
}

function renderExamples(state, api) {
  const engines = ['unity', 'unreal', 'godot', 'physis'];
  const subTabs = engines.map(e => {
    const has = api.examples && api.examples[e];
    return `<button class="ex-engine-tab ${state.modalExampleEngine === e ? 'active' : ''}"

}

// src/codeblock.js - 轻量级语法着色（不引入 highlighter 库）
const TOKEN_RULES = {
  csharp: [
    [/\b(var|void|int|float|bool|string|class|public|private|static|new|return|if|else|for|while)\b/g, 'kw'],
    [/("[^"]*")/g, 'str'],
    [/\/\/.*$/gm, 'cm'],
    [/\b(\d+\.?\d*)\b/g, 'num']
  ],
  cpp: [
    [/\b(void|int|float|bool|char|class|public|private|static|const|new|delete|return|if|else|for|while|true|false|auto)\b/g, 'kw'],
    [/("[^"]*")/g, 'str'],
    [/\/\/.*$/gm, 'cm'],
    [/(\b\w+(?=\s*\())/g, 'fn']
  ],
  gdscript: [
    [/\b(var|func|class|extends|if|elif|else|for|while|return|true|false|null|onready|export)\b/g, 'kw'],
    [/("[^"]*")/g, 'str'],
    [/(#[^!].*$)/gm, 'cm'],
    [/(\b\d+\.?\d*)\b/g, 'num']
  ],
  c: [
    [/\b(void|int|float|double|char|long|short|unsigned|signed|const|static|struct|return|if|else|for|while|sizeof|typedef)\b/g, 'kw'],
    [/("[^"]*")/g, 'str'],
    [/\/\/.*$/gm, 'cm'],
    [/\/\*[\s\S]*?\*\//g, 'cm'],
    [/(#\s*include\b)/g, 'pp'],
    [/(\b\w+(?=\s*\())/g, 'fn']
  ]
};

export function renderCodeBlock(language, code) {
  const rules = TOKEN_RULES[language] || TOKEN_RULES.c;
  let html = escapeHtml(code);
  // 简单替换：用 span 包裹；注意：先 cm 再 kw 避免注释内的关键字被着色
  for (const [regex, cls] of rules) {
    html = html.replace(regex, (m) => `<span class="tok-${cls}">${m}</span>`);
  }
  return `<pre class="code code-${language}"><code>${html}</code></pre>`;
}
```

> **实现说明**：上述替换在 Escape 后进行，避免对 `<` `>` `&` 的误伤；用顺序保证注释优先于关键字。

// src/keyboard.js
export function installGlobalKeys() {
  document.addEventListener('keydown', (e) => {
    const tag = e.target.tagName;
    if (tag === 'INPUT' || tag === 'TEXTAREA') return;

    if (e.key === '/') {
      e.preventDefault();
      document.getElementById('search-input')?.focus();
    }
  });

  // 列表导航
  document.getElementById('api-list')?.addEventListener('keydown', (e) => {
    const items = [...e.currentTarget.querySelectorAll('.api-item')];
    const idx = items.indexOf(document.activeElement);
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      items[Math.min(idx + 1, items.length - 1)]?.focus();
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      items[Math.max(idx - 1, 0)]?.focus();
    } else if (e.key === 'Enter' && idx >= 0) {
      items[idx].click();
    }
  });
}

## 4. 主入口（main.js）

```js
// src/main.js
import { loadAll } from './data-loader.js';
import { getState, setState, subscribe } from './state.js';
import { renderEngineTabs, renderCategoryTabs } from './tabs.js';
import { renderList } from './list.js';
import { renderModal, openModal, closeModal } from './modal.js';
import { installGlobalKeys } from './keyboard.js';

async function bootstrap() {
  const banner = document.getElementById('banner');
  try {
    const { data, categories, i18n } = await loadAll();
    setState({ data, categories, i18n });
  } catch (e) {
    banner.textContent = getState().i18n?.strings?.['data.loadError']?.replace('{error}', e.message)
      ?? `数据加载失败: ${e.message}`;
    banner.classList.add('error');
    return;
  }

  installGlobalKeys();

  // 渲染各区域
  subscribe(() => renderAll());

  // 事件绑定
  document.getElementById('engine-tabs').addEventListener('click', (e) => {
    const t = e.target.closest('[data-engine]');
    if (!t) return;
    setState({ engine: t.dataset.engine, category: 'all' });
  });

  document.getElementById('cat-tabs').addEventListener('click', (e) => {
    const t = e.target.closest('[data-category]');
    if (!t) return;
    setState({ category: t.dataset.category });
  });

  document.getElementById('search-input').addEventListener('input', (e) => {
    setState({ search: e.target.value });
  });

  document.getElementById('api-list').addEventListener('click', (e) => {
    const item = e.target.closest('[data-api-id]');
    if (item) openModal(item.dataset.apiId);
  });

  document.getElementById('modal-root').addEventListener('click', (e) => {
    if (e.target.dataset.close === '1') closeModal();
    const tab = e.target.closest('[data-modal-tab]');
    if (tab) setState({ modalTab: tab.dataset.modalTab });
    const ex = e.target.closest('[data-ex-engine]');
    if (ex && !ex.disabled) setState({ modalExampleEngine: ex.dataset.exEngine });
  });

  setState({}); // 触发首次渲染
}

function renderAll() {
  const s = getState();
  renderEngineTabs(s, document.getElementById('engine-tabs'));
  renderCategoryTabs(s, document.getElementById('cat-tabs'));
  renderList(s, document.getElementById('api-list'));
  renderModal(s, document.getElementById('modal-root'));
  document.getElementById('list-count').textContent =
    s.i18n.strings['list.count'].replace('{n}', document.querySelectorAll('.api-item').length);
}

bootstrap();
```

## 5. 初始数据样例（v0.1 录入 8 条）

为了验证流程，v0.1 录入以下 8 条 API（每引擎 2 条）：

| id | engine | name | category | 用途 |
|----|--------|------|----------|------|

| `unity-particlesystem-shape` | unity | `ParticleSystem.shape` | shape | 发射器形状 |
| `unreal-niagara-component-activate` | unreal | `UNiagaraComponent::Activate` | spawn | 启动发射 |
| `unreal-niagara-spawn-burst` | unreal | `Spawn Burst Instantaneous` | spawn | 瞬时爆发 |
| `godot-gpu-particles-emitting` | godot | `GPUParticles3D.emitting` | emission | 发射开关 |
| `godot-cpu-particles-amount` | godot | `CPUParticles3D.amount` | emission | 粒子数量 |
| `physis-gvpe-particle-emit` | physis | `gvpe_particle_emit` | emission | 同步发射 |
| `physis-gvpe-particle-create` | physis | `gvpe_particle_create` | lifecycle | 创建发射器 |

每条数据按 §1.2 的 schema 完整填写 4 引擎示例（v0.1 中部分示例可能标"参考实现"占位，正式 v0.2 由人工 review）。
